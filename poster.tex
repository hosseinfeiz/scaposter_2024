% ------------------------- Introduction ------------------------
\block[titleoffsety=0.25cm,bodyoffsety=0.25cm]{Introduction}{
  \fontsize{36}{40}\selectfont
  Motion capture in sports scenes like boxing presents challenges due to
  occlusion, crowding, and fast movements. We use a multi-view setup that
  requires visibility from only two cameras. Our pipeline produces realistic
  motions through multi-person physics optimisation while accounting for
  inter-character collisions.
}

  \vspace{1em}
  \fontsize{36}{42}\selectfont
  The pipeline begins with bounding boxes ($\mathbf{bboxes}$) and robust
  tracking IDs. 2D poses ($\mathbf{J}_\text{2D}$) from ViTPose~\cite{xu2022vitpose}
  are triangulated to 3D ($\mathbf{J}_\text{3D}$). Kinematics optimisation fits
  SMPL ($\theta,\beta$), then an iLQR physics stage \cite{howell2022} corrects
  artefacts using state $(\mathbf{q},\mathbf{v},\mathbf{p})$.
}

% ------------- Multi-frame / Multi-view tracking IDs -----------
\block[titleoffsetx=0cm,bodyoffsetx=0cm,
       titleoffsety=1cm,bodyoffsety=1cm]{Multi-frame Multi-view Tracking IDs}{
  \begin{itemize}\fontsize{29}{34}\selectfont
    \item \textbf{Tracking by segmentation}: XMem memory for stable IDs \cite{cheng2022xmem}
    \item \textbf{Epipolar ID matching}: Centroid \& epipolar constraints \cite{wohler20093d}
    \item \textbf{Top-down 2D poses}: ViTPose inside each box \cite{xu2022vitpose}
  \end{itemize}
}

% -------------- Weighted triangulation & filtering -------------
\block[titleoffsetx=0cm,bodyoffsetx=0cm,
       titleoffsety=1cm,bodyoffsety=1cm]{Weighted Triangulation and Filtering}{
  \fontsize{38}{40}\selectfont
  Weighted triangulation solves
  \[
    \mathbf{A}\,
    \begin{bmatrix}x\\y\\z\\1\end{bmatrix}=0,\qquad
    \mathbf{A}\!=\!
    \begin{bmatrix}
      \mu_1(P_{11}-u_1P_{31})\\
      \mu_1(P_{21}-v_1P_{31})\\
      \vdots\\
      \mu_N(P_{1N}-u_NP_{31N})\\
      \mu_N(P_{2N}-v_NP_{32N})
    \end{bmatrix}.
  \]
  SVD yields $(x,y,z)$. Cubic-spline smoothing removes outliers; an EKF fills
  missing frames when triangulation fails.
}

% -------------------- Kinematics optimisation ------------------
\block[titleoffsety=0.25cm,bodyoffsety=0.25cm]{Kinematics Optimisation}{
  \fontsize{38}{40}\selectfont
  Optimise SMPL $\theta$ by
  \[
    \min_{\theta}\!
      \sum\nolimits_t(
        w_1L_{2D}+w_2L_{3D}+w_3L_{\text{reg}}+
        w_4L_{\text{smooth}}+w_5L_{\text{GMM}}+w_6L_{\text{Vposer}}).
  \]
}

% ---------------- Physics body model generator -----------------
\useblockstyle{blockred}
\block{Physics Body Model Generator}{
  \fontsize{38}{40}\selectfont
  For $K$ humanoids the state is
  $\mathbf{x}_t=(\mathbf{q}_t,\mathbf{v}_t)\in\mathbb{R}^{125K}$,
  with $\mathbf{p}_t\in\mathbb{R}^{21K}$ SMPL joints. We export to
  MuJoCo XML \cite{todorov2012mujoco}.
}

% ---------------- Multi-person dynamics opt --------------------
\useblockstyle{feiz}
\block[titleoffsetx=0cm, bodyoffsetx=0cm, titleoffsety=1.3cm, bodyoffsety=1.3cm]{Multi-person Dynamics Optimization}{
\vspace{-0.39\baselineskip}
\fontsize{38}{40}\selectfont
The iLQR algorithm~\cite{howell2022} optimizes a control trajectory \(\mathbf{u}_{0:T}\) by minimizing the cost function. The reference positions. $\mathbf{p}_{t}$ and velocities $\mathbf{v}_t$ are extracted from the kinematics optimization at each frame $t$, and the trajectory optimization is "warm-started" using control parameters $\mathbf{u}_{t-1}$ from the previous frame. The control is updated iteratively as \(\Delta\mathbf{u}_t = \mathbf{K}_t\Delta\mathbf{x}_t + \alpha \mathbf{k}_t\), with \(\mathbf{K}_t\) and \(\mathbf{k}_t\) refined until convergence.
\vspace{-0.3\baselineskip}
\fontsize{30}{36}\selectfont
\innerblock[titlewidthscale=1, bodywidthscale=1]{Dynamics Optimization Formulation: $ \min\limits_{\mathbf{u}_{0:T}} \sum\limits_{t\in [0,T]} w_1 \mathrm{L_{\text{reg},t}} + w_2 \mathrm{L}_{\text{p},t} + w_3 \mathrm{L}_{\text{v},t} + w_4 \mathrm{L}_{\text{collision},t}$}{
\begin{multicols}{2}
    \setlength{\itemsep}{0pt} % Remove space between items
\begin{itemize}
    \item \textbf{Position Loss}: Minimizes the difference between predicted and actual 3D joint positions in each short horizon. \\
    $\mathrm{L}_{\text{p},t} = \sum\limits_{\text{i} \in \text{$\mathbf{J}_\text{3D}$}} \| \mathrm{p}_\text{i,t} - \mathrm{J}_\text{3D,i,t}\|^2$  
    \vspace{-0.4\baselineskip}
    \item \textbf{Velocity Loss}: Minimizes the difference between predicted and actual 3D joint velocities in each short horizon.\\  
    $\mathrm{L}_{\text{v},t} = \sum\limits_{\text{i} \in \text{$\mathbf{J}_{3D}$}} c_i \| \mathrm{v}_\text{i,t} - \mathrm{v}_{3D,i,t}\|^2$
\end{itemize}
    \begin{itemize}
        \item \textbf{Collision Loss}: The collision term penalizes the penetration between the geoms of different humanoids.  \\
        $\mathrm{L}_{\text{collision},t} = \sum\limits_{\text{i\in{1\dots M}}} \sum\limits_{\text{j\in{1\dots M}}} \max(0, \epsilon - \text{dist}(G_{1_i}, G_{2_j}))$
        \item \textbf{Regularization Loss}: Prevent sudden changes in control input. $\mathrm{L}_{\text{reg},t} = \|\mathbf{u}_t\|^2$
    \end{itemize}
\end{multicols}
\vspace{-0.39\baselineskip}
}
\vspace{-0.5\baselineskip}
